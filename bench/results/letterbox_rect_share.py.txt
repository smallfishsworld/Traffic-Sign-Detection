"""Share of the fixed 640x640 letterbox's pixels that Ultralytics' minimal-rectangle letterbox
(long side 640, short side padded to a multiple of 32) would use, on the 16 benchmark images."""
import math
from pathlib import Path

from PIL import Image

from tsd.benchmark import pick_images

paths = pick_images(sorted(Path("data/openimages-traffic/images").rglob("*.jpg")), 16)
ratios = []
for p in paths:
    w, h = Image.open(p).size
    r = 640 / max(w, h)
    pw, ph = math.ceil(round(w * r) / 32) * 32, math.ceil(round(h * r) / 32) * 32
    ratios.append(pw * ph / 640 / 640)
    print(p.name, w, h, pw, ph, round(ratios[-1], 3))
print("images", len(paths), "mean", round(sum(ratios) / len(ratios), 3), "min", round(min(ratios), 3), "max", round(max(ratios), 3))
